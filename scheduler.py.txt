# ตัวตั้งเวลาให้ทำงานเองทุกวัน - ใช้กับ Render แบบ Background Worker
import schedule
import time
import os
from app import run_one_episode

TOPICS = [
    "น้องมะขามหลงป่า",
    "น้องมะขามช่วยเพื่อนเก็บมะขาม",
    "น้องมะขามไปตลาดครั้งแรก",
    "น้องมะขามกลัวฟ้าร้อง",
    "น้องมะขามทำขนมให้เพื่อน",
]
day = 0
def job():
    global day
    topic = TOPICS[day % len(TOPICS)]
    day += 1
    print(f"ถึงเวลาแล้ว ทำงาน EP: {topic}")
    run_one_episode(topic)

# ตั้งเวลา 06:00 ทุกวัน (เวลาไทย ต้องตั้งเป็น 23:00 UTC ของวันก่อนหน้าใน Render)
schedule.every().day.at("23:00").do(job)

print("ระบบถาวรเริ่มทำงานแล้ว รอถึงเวลา...")
# รันทันที 1 ครั้งเพื่อทดสอบ
job()
while True:
    schedule.run_pending()
    time.sleep(60)
